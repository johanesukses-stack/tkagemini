Panduan GitHub & Apps Script

Upload file TKA_GURU.html ke repository GitHub Pages Anda sebagai index.html. Siswa yang membuka link dengan parameter ?mode=siswa&pkg=... akan otomatis diarahkan ke halaman ujian online, dan nilai akan terkirim ke Google Spreadsheet Anda.

// Google Apps Script Code.gs snippet
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Nilai_Siswa') || ss.insertSheet('Nilai_Siswa');
  sheet.appendRow([new Date(), data.studentName, data.studentClass, data.subject, data.score, data.correct, data.wrong]);
  return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
}
